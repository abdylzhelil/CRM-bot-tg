async def test_observed_local_manual_query_delayed_until_russia_tier(maker):
    async with maker() as s:
        s.add(SearchQuery(text='ремонт квартир Каспийск',enabled=True))
        await s.commit()
        tiers=await search.tier_queries(s)
        assert 'ремонт квартир Каспийск' not in tiers[0]
        assert 'ремонт квартир Каспийск' in tiers[2]
        batch,stage,_=await search.next_batch(s)
        assert stage==0 and 'ремонт квартир Каспийск' not in batch

async def test_observed_failed_initial_attach_never_retried_by_watchdog(maker,monkeypatch):
    async with maker() as s:
        acc=ScoutAccount(id=98,name='synthetic retry',api_id=1,api_hash='test',session='synthetic')
        s.add(acc);await s.commit()
    client=SimpleNamespace(connect=AsyncMock(side_effect=ConnectionError('synthetic outage')))
    monkeypatch.setattr(manager_mod,'make_client',lambda acc:client)
    monkeypatch.setattr(manager_mod.notify,'alert',AsyncMock())
    m=manager_mod.ScoutManager()
    await m.attach(acc)
    assert not m.clients
    calls=0
    async def tick(delay):
        nonlocal calls
        calls+=1
        if calls>1:raise asyncio.CancelledError()
    monkeypatch.setattr(manager_mod.asyncio,'sleep',tick)
    with pytest.raises(asyncio.CancelledError):await m.watchdog()
    assert client.connect.await_count==1

async def test_observed_two_manual_or_background_searches_overlap(monkeypatch):
    entered=0
    max_active=0
    release=asyncio.Event()
    async def operation(client,report=False):
        nonlocal entered,max_active
        entered+=1
        max_active=max(max_active,entered)
        if entered==2:release.set()
        await release.wait()
        entered-=1
        return 0
    monkeypatch.setattr(search,'_discover',operation)
    await asyncio.wait_for(asyncio.gather(search.discover(object()),search.discover(object())),2)
    assert max_active==2
